\section{Proof of Theorem~\ref{thm:ch02-regret_bound}}
\label{app:ch02-proof_thm1}

Let $\gX_{\on}, \gX_{\off}$ be an arbitrary (not necessarily disjoint) partition of $\gX = \gS \times \gA \times [H]$. We bound the regret for this fixed partition, which immediately yields the infimum over all partitions in Theorem~\ref{thm:ch02-regret_bound}.

Following \citet{xie2022role}, we write $\delta_h^{(t)} \coloneqq f_h^{(t)} - \gT_h f_{h+1}^{(t)}$ for the Bellman residual at online episode $t$ and step $h$, and define the cumulative in-sample occupancy measures (without and with the offline dataset) by
\begin{equation}
    \bar{d}_h^{(t)} \coloneqq \sum_{i = 1}^{t-1} d^{(i)}_h, \qquad \tilde{d}_h^{(t)} \coloneqq  N_{\off} \mu_{h} +  \sum_{i = 1}^{t-1} d_h^{(i)},
\end{equation}
where $d^{(t)}_h = d^{\pi^{(t)}}_h$ is the occupancy measure induced by the greedy policy $\pi^{(t)} = \pi_{f^{(t)}}$. We further write
\begin{equation}
    \E_{\bar d_h^{(t)}}[g] = \sum_{i = 1}^{t-1} \E_{d_h^{(i)}}[g], \qquad \E_{\tilde{ d}_h^{(t)}}[g] = \sum_{i = 1}^{t-1} \E_{d_h^{(i)}}[g] + N_{\off} \E_{\mu_{h}}[g].
\end{equation}

We first state the high-probability confidence-set and in-sample squared Bellman error guarantee, which extends Lemma 15 of \citet{xie2022role} and Lemma 44 of \citet{jin2021bellmaneluder} to include the $N_{\off}$ offline episodes.

\begin{lem}[High-Probability Confidence Set and In-Sample Error]
\label{lem:ch02-good_event}
With probability at least $1-\delta$, for all $t \in [N_{\on}]$ and all $h \in [H]$,
\begin{equation}
    \text{(i) } Q^\star_h \in \mathcal{F}^{(t)}_h, \qquad \text{(ii) } \E_{\tilde{d}_h^{(t)}}\!\left[(\delta_h^{(t)})^2\right] \leq \gO(H^2\beta),
\end{equation}
where $\beta = c_1\left(\log \left[N H \mathcal{N}_{\mathcal{F}}(\rho) / \delta\right]+ N \rho\right)$ for an absolute constant $c_1 > 0$ and $N = N_{\off} + N_{\on}$.
\end{lem}

\begin{proof}
Lemma 44 of \citet{jin2021bellmaneluder} establishes that for any sequence of $N$ episodes, with probability at least $1-\delta$, (i) the optimal value function $Q^\star$ belongs to the confidence set at every episode $\tau \in [N]$, and (ii) any function $\bar{f}^{(\tau)}$ in the confidence set at episode $\tau$ has cumulative expected squared Bellman error $\sum_{i=1}^{\tau-1} \E_{d_h^{(i)}}[(\bar{f}_h^{(\tau)} - \gT_h \bar{f}_{h+1}^{(\tau)})^2] \leq \gO(H^2 \beta)$ over the distributions generating the first $\tau - 1$ episodes. Ordering the $N_{\off}$ offline episodes arbitrarily as episodes $\tau = 1, \dots, N_{\off}$ (each drawn from $\mu$) and indexing the $N_{\on}$ online episodes as $\tau = N_{\off} + 1, \dots, N_{\off} + N_{\on}$ with $\bar{f}^{(N_{\off} + t)} = f^{(t)} \in \gF^{(t-1)}$, the conclusion holds simultaneously for all $\tau \in [N]$, and in particular for all online episodes $t \in [N_{\on}]$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:ch02-regret_bound}]
Conditioned on the good event of Lemma~\ref{lem:ch02-good_event} (so that $Q^\star \in \gF^{(t-1)}$ and optimism $f_1^{(t)}(s_1^{(t)}, \pi_1^{(t)}(s_1^{(t)})) \geq V_1^\star(s_1^{(t)})$ holds for every $t \in [T]$), the standard performance-difference regret decomposition \citep[Lemma 3]{xie2022role} yields
\begin{equation}
    \Reg(T) \leq \sum_{t = 1}^T \sum_{h=1}^H \E_{(s, a) \sim d_h^{(t)}} \!\left[\delta_h^{(t)}(s, a)\right]. \label{equ:ch02-reg_decp}
\end{equation}
Using the partition $\gX_{\on} \cup \gX_{\off} = \gX$ and writing $\delta_{h, \on}^{(t)}(s, a) \coloneqq \delta_{h}^{(t)}(s, a) \mathbbm{1}_{(s, a, h) \in \gX_{\on}}$ and $\delta_{h, \off}^{(t)}(s, a) \coloneqq \delta_{h}^{(t)}(s, a) \mathbbm{1}_{(s, a, h) \in \gX_{\off}}$, Equation~\eqref{equ:ch02-reg_decp} decomposes as
\begin{equation}
    \Reg(T) \leq \sum_{t = 1}^T \sum_{h=1}^H \E_{(s, a) \sim d_h^{(t)}} \!\left[\delta_{h, \on}^{(t)}(s, a)\right] + \sum_{t = 1}^T \sum_{h=1}^H \E_{(s, a) \sim d_h^{(t)}} \!\left[\delta_{h, \off}^{(t)}(s, a)\right].
\end{equation}

For the online partition term, applying the Cauchy--Schwarz inequality and Lemma~\ref{lem:ch02-good_event} gives
\begin{align}
    \sum_{t = 1}^T \sum_{h=1}^H \E_{d_h^{(t)}} \!\left[\delta_{h, \on}^{(t)}\right]
    &= \sum_{t = 1}^T \sum_{h=1}^H \E_{d_h^{(t)}} \!\left[\delta_{h, \on}^{(t)} \left(\frac{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]}\right)^{1 / 2} \right] \nonumber\\
    &\leq \sqrt{\sum_{t = 1}^T \sum_{h=1}^H \frac{\E_{d_h^{(t)}}[\delta_{h, \on}^{(t)}(s, a)]^2}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]} } 
    \sqrt{\sum_{t = 1}^T \sum_{h=1}^H\left(H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]\right)} \nonumber\\
    &\leq \sqrt{H \cdot c_{\on}(\gF, \gX_{\on}, T)} \cdot \sqrt{HT \cdot \gO(H^2\beta)} \nonumber\\
    &\lesssim H^2\sqrt{\beta \cdot c_{\on}(\gF, \gX_{\on}, T) \cdot T}.
\end{align}

For the offline partition term, we similarly apply the Cauchy--Schwarz inequality with respect to the combined in-sample measure $\tilde{d}_h^{(t)}$:
\begin{align}
    \sum_{t = 1}^T \sum_{h=1}^H \E_{d_h^{(t)}} \!\left[\delta_{h, \off}^{(t)}\right]
    &= \sum_{t = 1}^T \sum_{h=1}^H \E_{(s, a) \sim d_h^{(t)}}\!\left[\delta_{h, \off}^{(t)}(s, a)\left(\frac{ N_{\off} \E_{\mu_{h}}[(\delta_{h, \off}^{(t)})^2] + \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \off}^{(t)})^2]}{ N_{\off} \E_{\mu_{h}}[(\delta_{h, \off}^{(t)})^2] + \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \off}^{(t)})^2]}\right)^{1 / 2}\right]  \nonumber\\
    &\leq \sqrt{\sum_{t = 1}^T \sum_{h=1}^H \frac{\E_{(s, a) \sim d_h^{(t)}}\!\left[\left( \delta_{h, \off}^{(t)}(s, a)\right)^2\right]}{ N_{\off} \E_{\mu_{h}}[(\delta_{h, \off}^{(t)})^2] + \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \off}^{(t)})^2]}} \nonumber\\
    &\qquad \cdot\sqrt{\sum_{t=1}^T \sum_{h=1}^H  \left(  N_{\off} \E_{\mu_{h}}[(\delta_{h, \off}^{(t)})^2] + \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \off}^{(t)})^2]\right)}.
\end{align}
By definition of $c_{\off}(\gF, \gX_{\off})$, the first factor is bounded by
\begin{align}
    \sqrt{\sum_{t = 1}^T \sum_{h=1}^H \frac{\E_{d_h^{(t)}}[( \delta_{h, \off}^{(t)})^2]}{ N_{\off} \E_{\mu_{h}}[(\delta_{h, \off}^{(t)})^2] + \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \off}^{(t)})^2]}}
    &\leq \sqrt{\sum_{t=1}^T \sum_{h=1}^H \frac{\E_{d_h^{(t)}}[( \delta_{h, \off}^{(t)})^2]}{ N_{\off} \E_{\mu_{h}}[(\delta_{h, \off}^{(t)})^2]}}  \nonumber\\
    &\leq \sqrt{\frac{HT}{N_{\off}} c_{\off}(\gF, \gX_{\off})}.
\end{align}
By Lemma~\ref{lem:ch02-good_event}, the second factor is bounded by
\begin{equation}
    \sqrt{\sum_{t=1}^T \sum_{h=1}^H \E_{\tilde{d}_h^{(t)}}[(\delta_h^{(t)})^2]} \lesssim H\sqrt{H T \beta}.
\end{equation}
Combining the online and offline bounds and setting $\rho = 1/N$ yields Theorem~\ref{thm:ch02-regret_bound}.
\end{proof}

\section{Bounding the Restricted Sequential Extrapolation Coefficient}
\label{app:ch02-sec_proofs}

To instantiate Theorem~\ref{thm:ch02-regret_bound} in specific MDP families, we relate the restricted sequential extrapolation coefficient $c_{\on}(\gF, \gX_{\on}, T)$ to restricted coverability and the Distributional-Eluder dimension \citep{jin2021bellmaneluder, xie2022role}.

\begin{defn}[Distributional-Eluder Dimension]
\label{defn:ch02-DE_dim}
The Distributional-Eluder dimension $\dim_{\operatorname{DE}}(\gG, \sD, \epsilon)$ of a function class $\gG$ with respect to a distribution family $\sD$ at scale $\epsilon > 0$ is the length $n \in \sN$ of the longest sequence $\{d^{(1)}, \dots, d^{(n)}\} \subset \sD$ and $\{g^{(1)}, \dots, g^{(n)}\} \subset \gG$ such that for all $t \in [n]$, there exists $\varepsilon^{(t)} \geq \epsilon$ with
\begin{equation}
    \left|\mathbb{E}_{d^{(t)}}\!\left[g^{(t)}\right]\right| > \varepsilon^{(t)} \quad \text{and} \quad \sqrt{\sum_{i=1}^{t-1}\left(\mathbb{E}_{d^{(i)}}\!\left[g^{(t)}\right]\right)^2} \leq \varepsilon^{(t)}.
\end{equation}
\end{defn}

\begin{lem}[Restricted Per-State-Action Elliptic Potential Lemma]
\label{lem:ch02-restricted-elliptic}
Consider any sequence of occupancy densities $d_h^{(1)},\dots,d_h^{(T)}$ and a partition $\gX_{\off}, \gX_{\on}$ of $\gX$. Define
\begin{equation}
    \mu^\star_h \coloneqq \argmin_{\mu_h \in \Delta(\gS \times \gA)} \sup_{\pi \in \Pi} \frac{\E_{(s, a) \sim d_h^{\pi}}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}{\E_{(s, a) \sim \mu_h}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}, \qquad
    c_h(\gX_{\on}) \coloneqq \inf_{\mu_h \in \Delta(\gS \times \gA)} \sup_{\pi \in \Pi} \frac{\E_{(s, a) \sim d_h^{\pi}}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}{\E_{(s, a) \sim \mu_h}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}.
\end{equation}
Then for all $(s,a,h)\in \gX_{\on}$, we have $d_h^{(t)}(s,a) / \mu^\star_h(s,a) \leq c_h(\gX_{\on})$ and
\begin{equation}
    \sum_{t = 1}^T\frac{d_h^{(t)}(s,a)\mathbbm{1}((s,a,h) \in \gX_{\on})}{\left(c_h(\gX_{\on})\cdot\mu_h^\star(s,a) + \sum_{i=1}^{t-1} d_h^{(i)}(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})} \leq 2\log(1+T).
\end{equation}
\end{lem}

\begin{proof}
Since $d_h^{(t)}(s,a) \leq c_h(\gX_{\on}) \mu_h^\star(s,a)$ on $\gX_{\on}$, each summand lies in $[0, 1]$. Using $u \leq 2\log(1+u)$ for $u \in [0, 1]$ and telescoping the product over $t = 1, \dots, T$, we obtain
\begin{align}
    &\sum_{t = 1}^T\frac{d_h^{(t)}(s,a)\mathbbm{1}((s,a,h) \in \gX_{\on})}{\left(c_h(\gX_{\on})\cdot\mu_h^\star(s,a) + \sum_{i=1}^{t-1} d_h^{(i)}(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})} \nonumber\\
    &\quad\leq 2\sum_{t = 1}^T\log\!\left(\frac{\left(c_h(\gX_{\on})\cdot\mu_h^\star(s,a) + \sum_{i=1}^{t} d_h^{(i)}(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})}{\left(c_h(\gX_{\on})\cdot\mu_h^\star(s,a) + \sum_{i=1}^{t-1} d_h^{(i)}(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})}\right) \nonumber\\
    &\quad= 2\log\!\left(\frac{\left(c_h(\gX_{\on})\cdot\mu_h^\star(s,a) + \sum_{i=1}^T d_h^{(i)}(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})}{\left(c_h(\gX_{\on})\cdot\mu_h^\star(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})}\right) \leq 2\log(1+T).
\end{align}
\end{proof}

\begin{lem}[Bounds on Restricted SEC, Modified from Propositions 13 and 14 of \citet{xie2022role}]
\label{lem:ch02-general_SEC}
For any partition $\gX_{\off}, \gX_{\on}$ of $\gX = \gS \times \gA \times [H]$, the restricted online complexity measure $c_{\on}(\gF, \gX_{\on}, T)$ satisfies:
\begin{align}
    \text{(1)} \quad c_{\on}(\gF, \gX_{\on}, T) &\lesssim \log(T) \max_{h \in [H]} \inf_{\mu_h \in \Delta(\gS \times \gA)} \sup_{\pi \in \Pi} \frac{\E_{(s, a) \sim d_h^{\pi}}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}{\E_{(s, a) \sim \mu_h}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}, \\
    \text{(2)} \quad c_{\on}(\gF, \gX_{\on}, T) &\lesssim \inf_{\epsilon > 0}\left\{\epsilon^2T + \max_{h \in [H]} \dim_{\operatorname{DE}}\!\left((\gF_h - \gT_h \gF_{h+1})\mathbbm{1}_{(\cdot, h) \in \gX_{\on}}, \sD_h, \epsilon\right)\right\}\log(T).
\end{align}
\end{lem}

\begin{proof}
We first prove (1). Fix $h \in [H]$ and arbitrary sequences $f^{(1)}, \dots, f^{(T)} \in \gF$ and $\pi^{(1)}, \dots, \pi^{(T)} \in \Pi$, writing $d_h^{(t)} = d_h^{\pi^{(t)}}$ and $\tilde{d}_h^{(t)} = \sum_{i=1}^{t-1} d_h^{(i)}$. Define the per-state-action stopping time
\begin{equation}
    \tau(s,a) \coloneqq \min\left\{t \in [T] : \sum_{i=1}^{t-1} d_h^{(i)}(s,a) \geq \mu^\star_h(s,a)\cdot c_h(\gX_{\on})\right\}.
\end{equation}
Decomposing $\E_{d_h^{(t)}}[\delta_{h,\on}^{(t)}(s,a)]$ into the burn-in indicator $\mathbbm{1}(t < \tau(s,a))$ and post-burn-in indicator $\mathbbm{1}(t \geq \tau(s,a))$, we have
\begin{equation}
    \sum_{t = 1}^T \frac{\E_{d_h^{(t)}} [\delta_{h, \on}^{(t)}(s, a)]^2}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]}
    \lesssim \sum_{t = 1}^T \frac{\E_{d_h^{(t)}} [\delta_{h, \on}^{(t)}(s, a)\mathbbm{1}(t < \tau(s,a))]^2}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]} + \sum_{t = 1}^T \frac{\E_{d_h^{(t)}} [\delta_{h, \on}^{(t)}(s, a)\mathbbm{1}(t \geq \tau(s,a))]^2}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]}.
\end{equation}
Since $|\delta_{h,\on}^{(t)}| \leq H$, the burn-in sum is bounded by
\begin{align}
    \sum_{t=1}^T \E_{d_h^{(t)}}\!\left[\mathbbm{1}(t < \tau(s,a))\mathbbm{1}((s,a,h) \in \gX_{\on})\right]
    &= \int_{\gS \times \gA} \left(\tilde{d}_h^{(\tau(s,a)-1)}(s,a) + d_h^{(\tau(s,a)-1)}(s,a)\right)\mathbbm{1}((s,a,h) \in \gX_{\on})\,\mathrm{d}(s,a) \nonumber\\
    &\leq 1 + c_h(\gX_{\on}) \int_{\gS \times \gA} \mu_h^\star(s,a)\mathbbm{1}((s,a,h) \in \gX_{\on})\,\mathrm{d}(s,a) \leq 1 + c_h(\gX_{\on}).
\end{align}
For the post-burn-in sum, Cauchy--Schwarz gives
\begin{equation}
    \frac{\E_{d_h^{(t)}}[\delta_{h, \on}^{(t)}(s,a)\mathbbm{1}(t \geq \tau(s,a))]^2}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]}
    \leq \int_{\gS \times \gA}\frac{d_h^{(t)}(s,a)^2\mathbbm{1}(t \geq \tau(s,a))\mathbbm{1}((s,a,h) \in \gX_{\on})}{\sum_{i=1}^{t-1} d_h^{(i)}(s,a)\mathbbm{1}((s,a,h) \in \gX_{\on})}\,\mathrm{d}(s,a).
\end{equation}
Using $\sum_{i=1}^{t-1} d_h^{(i)}(s,a) \geq \frac{1}{2}(c_h(\gX_{\on})\mu_h^\star(s,a) + \sum_{i=1}^{t-1} d_h^{(i)}(s,a))$ for $t \geq \tau(s,a)$, bounding one factor of $d_h^{(t)}(s,a)$ in the numerator by $c_h(\gX_{\on})\mu_h^\star(s,a)$, and applying Lemma~\ref{lem:ch02-restricted-elliptic} yields
\begin{align}
    \sum_{t = 1}^T \frac{\E_{d_h^{(t)}} [\delta_{h, \on}^{(t)}(s, a)\mathbbm{1}(t \geq \tau(s,a))]^2}{H^2 \vee \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[(\delta_{h, \on}^{(t)})^2]}
    &\leq 2c_h(\gX_{\on})\int_{\gS \times \gA}\mu_h^\star(s,a)\sum_{t = 1}^T\frac{d_h^{(t)}(s,a)\mathbbm{1}((s,a,h) \in \gX_{\on})}{\left(c_h(\gX_{\on})\mu_h^\star(s,a) + \sum_{i=1}^{t-1} d_h^{(i)}(s,a)\right)}\,\mathrm{d}(s,a) \nonumber\\
    &\lesssim c_h(\gX_{\on})\log(T).
\end{align}
Taking the maximum over $h \in [H]$ proves (1).

To prove (2), let $\dimde(\epsilon) \coloneqq \dim_{\operatorname{DE}}((\gF_h - \gT_h \gF_{h+1})\mathbbm{1}_{(\cdot, h) \in \gX_{\on}}, \sD_h, \epsilon)$ and $\beta_h^{(t)} \coloneqq \sum_{i=1}^{t-1} \E_{d_h^{(i)}}[(\delta_{h,\on}^{(t)})^2]$. For any $\alpha \in (0, 1]$, let $i_1, \dots, i_k$ be the indices $t \in [T]$ where $\E_{d_h^{(t)}}[\delta_{h,\on}^{(t)}]^2 / (H^2 \vee \beta_h^{(t)}) > \alpha^2$. By the pigeonhole construction of \citet[Proposition 14]{xie2022role}, $d_h^{(i_k)}$ is $\alpha\sqrt{H^2 \vee \beta_h^{(i_k)}}$-dependent on at least $L^\star \geq \frac{k}{\dimde(\alpha)+1} - 1$ disjoint subsequences, whereas the definition of $\beta_h^{(i_k)}$ implies $L^\star \leq 1/\alpha^2$. Hence $k \leq \frac{3\dimde(\alpha)}{\alpha^2} + 1$. Ordering $e_h^{(t)} = \E_{d_h^{(t)}}[\delta_{h,\on}^{(t)}]^2 / (H^2 \vee \beta_h^{(t)})$ in decreasing order and integrating the tail bound over $t > \dimde(\sqrt{c})$ for $c = \epsilon^2$ yields
\begin{equation}
    \sum_{t=1}^T e_h^{(t)} \leq c T + \dimde(\sqrt{c}) + \sum_{t=\dimde(\sqrt{c})+1}^T \frac{6\dimde(\sqrt{c})}{t-1} \lesssim \left(\epsilon^2 T + \dimde(\epsilon)\right)\log(T),
\end{equation}
establishing (2).
\end{proof}

\subsection{Proof of Proposition~\ref{prop:ch02-case_tabular} (Tabular MDPs)}
By definition of $c_{\off}(\gF, \gX_{\off})$,
\begin{equation}
    c_{\off}(\gF, \gX_{\off}) = \max_{h \in [H]} \sup_{f \in \gF} \sup_{\pi \in \Pi} \frac{\E_{d_h^\pi}[(f_h - \gT_h f_{h+1})^2 \mathbbm{1}_{(\cdot, h) \in \gX_{\off}}]}{\E_{\mu_h}[(f_h - \gT_h f_{h+1})^2 \mathbbm{1}_{(\cdot, h) \in \gX_{\off}}]} \leq \sup_{\pi \in \Pi} \sup_{(s,a,h) \in \gX_{\off}} \frac{d_h^\pi(s,a)}{\mu_h(s,a)}.
\end{equation}
For the online complexity measure, applying Lemma~\ref{lem:ch02-general_SEC}(1) and taking $\mu_h$ to be the uniform distribution over $\gX_{\on, h}$ yields
\begin{equation}
    c_{\on}(\gF, \gX_{\on}, T) \lesssim \log(T) \max_{h \in [H]} \inf_{\mu_h \in \Delta(\gS \times \gA)} \sup_{\pi \in \Pi} \frac{\E_{(s, a) \sim d_h^{\pi}}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]}{\E_{(s, a) \sim \mu_h}[\mathbbm{1}_{(s, a, h) \in \gX_{\on}}]} \leq \log(T) \max_{h \in [H]} |\gX_{\on, h}|.
\end{equation}
Finally, for $\gF_h = [0, H]^{S A}$ and $\rho = 1/N$, we have $\log \gN_{\gF}(1/N) = \gO(H S A \log N)$. Substituting these bounds into Theorem~\ref{thm:ch02-regret_bound} completes the proof. \qed

\subsection{Proof of Proposition~\ref{prop:ch02-case_linear} (Linear MDPs)}
By Bellman completeness of linear MDPs \citep{jin2020provably}, for any $f_{h+1} \in \gF_{h+1}$, we have $f_h - \gT_h f_{h+1} = \langle \tilde{w}_h, \phi(\cdot, \cdot)\rangle$ with $\|\tilde{w}_h\|_2 \leq 2H\sqrt{d}$. Since $\gP_{\off}\phi(s,a) = \phi(s,a)$ for all $(s,a,h) \in \gX_{\off}$,
\begin{align}
    c_{\off}(\gF, \gX_{\off}) 
    &= \max_{h \in [H]} \sup_{\|\tilde w_h\|_2 \leq 2H\sqrt{d}} \sup_{\pi \in \Pi} \frac{\tilde w_{h}^\top \E_{d_h^{\pi}}\!\left[ \phi(s, a) \phi(s, a)^\top \mathbbm{1}_{(s, a, h) \in \gX_{\off}}\right] \tilde{w}_h}{\tilde w_{h}^\top \E_{\mu_h}\!\left[ \phi(s, a) \phi(s, a)^\top \mathbbm{1}_{(s, a, h) \in \gX_{\off}}\right] \tilde{w}_h} \nonumber\\
    &\leq \max_{h \in [H]} \sup_{\tilde w_h \in \Phi_{\off}} \sup_{\pi \in \Pi} \frac{\tilde w_{h}^\top \E_{d_h^{\pi}}\!\left[ \phi_{\off}(s, a) \phi_{\off}(s, a)^\top \right] \tilde w_{h}}{\tilde w_{h}^\top \E_{\mu_h}\!\left[ \phi_{\off}(s, a) \phi_{\off}(s, a)^\top \right] \tilde w_{h}} \nonumber\\
    &\leq \max_{h \in [H]} \frac{1}{\lambda_{d_{\off}}\!\left(\E_{\mu_h}\!\left[ \phi_{\off}(s, a) \phi_{\off}(s, a)^\top \right]\right)}.
\end{align}

To bound $c_{\on}(\gF, \gX_{\on}, T)$, let $V \in \sR^{d \times d_{\on}}$ be an orthonormal basis for $\Phi_{\on}$ so that $\phi(s,a) = V \phi_{\on}(s,a)$ with $\phi_{\on}(s,a) \in \sR^{d_{\on}}$ for all $(s,a,h) \in \gX_{\on}$. Then for any $g \in \gF_h - \gT_h \gF_{h+1}$ and $d \in \sD_h$, we can write $\E_{(s,a)\sim d}[g(s,a)\mathbbm{1}_{(s,a,h)\in \gX_{\on}}] = \langle \psi(g), \varphi(d)\rangle$ in $\sR^{d_{\on}}$ with $\|\psi(g)\|_2 \leq 2H\sqrt{d}$ and $\|\varphi(d)\|_2 \leq 1$. Suppose $d_1, \dots, d_m \in \sD_h$ is an $\epsilon$-independent sequence with witnesses $g_1, \dots, g_m$. Setting $\vx_i = \psi(g_i) \in \sR^{d_{\on}}$, $\vz_i = \varphi(d_i) \in \sR^{d_{\on}}$, and $\mV_i = \sum_{t=1}^{i-1} \vz_t \vz_t^\top + \frac{\epsilon^2}{4H^2 d}\mI_{d_{\on}}$, we have $\|\vx_i\|_{\mV_i} \leq \sqrt{2}\epsilon$ and $|\langle \vx_i, \vz_i\rangle| \geq \epsilon$, which implies $\|\vz_i\|_{\mV_i^{-1}}^2 \geq 1/2$. By the matrix determinant lemma and the AM--GM inequality on $\operatorname{tr}(\mV_m)$,
\begin{equation}
    \left(\frac{\epsilon^2}{4H^2 d}\right)^{d_{\on}}\left(\frac{3}{2}\right)^{m-1} \leq \det(\mV_m) \leq \left(\frac{m-1}{d_{\on}} + \frac{\epsilon^2}{4H^2 d}\right)^{d_{\on}},
\end{equation}
which implies $m = \dim_{\operatorname{DE}}((\gF_h - \gT_h \gF_{h+1})\mathbbm{1}_{(\cdot, h) \in \gX_{\on}}, \sD_h, \epsilon) \lesssim d_{\on} \log(H d / \epsilon)$. Applying Lemma~\ref{lem:ch02-general_SEC}(2) with $\epsilon = 1/\sqrt{T}$ gives $c_{\on}(\gF, \gX_{\on}, T) \lesssim d_{\on} \log(H T)\log(T)$. Combining this with $\log \gN_{\gF}(1/N) \lesssim d H \log(N H d)$ in Theorem~\ref{thm:ch02-regret_bound} proves Proposition~\ref{prop:ch02-case_linear}. \qed

\subsection{Proof of Proposition~\ref{prop:ch02-BMDP_bound} (Block MDPs)}
In a Block MDP (Definition~\ref{defn:ch02-block_MDP}), each state $s \in \gS$ uniquely determines its latent state $u = f^\star(s) \in \gU$. Pushing expectations forward from $\gS \times \gA$ to $\gU \times \gA$ under $f^\star$ gives
\begin{equation}
    c_{\off}(\gF, \gX_{\off}) \leq \sup_{\pi \in \Pi} \sup_{(u,a,h) \in \bar{\gX}_{\off}} \frac{\bar{d}_h^\pi(u,a)}{\bar{\mu}_h(u,a)}
\end{equation}
and, by Lemma~\ref{lem:ch02-general_SEC}(1),
\begin{equation}
    c_{\on}(\gF, \gX_{\on}, T) \lesssim \log(T) \max_{h \in [H]} \inf_{\bar{\mu}_h \in \Delta(\gU \times \gA)} \sup_{\pi \in \Pi} \frac{\E_{(u, a) \sim \bar{d}_h^{\pi}}[\mathbbm{1}_{(u, a, h) \in \bar{\gX}_{\on}}]}{\E_{(u, a) \sim \bar{\mu}_h}[\mathbbm{1}_{(u, a, h) \in \bar{\gX}_{\on}}]} \leq \log(T) \max_{h \in [H]} |\bar{\gX}_{\on, h}|.
\end{equation}
Substituting these bounds into Theorem~\ref{thm:ch02-regret_bound} proves Proposition~\ref{prop:ch02-BMDP_bound}. \qed

\section{Proof of Proposition~\ref{prop:ch02-general_recipe} (General Hybrid Recipe)}
\label{app:ch02-general_case}

\begin{proof}
By Condition 1, for any partition $\gX_{\on} \cup \gX_{\off} = \gX$,
\begin{equation}
    \Reg_{\gL}(T) \leq \sum_{t = 1}^T \sum_{h = 1}^H \E_{(s, a) \sim d^{(t)}_h}\!\left[\delta_h^{t}(s, a) \mathbbm{1}((s,a,h) \in \gX_{\on})\right] + \sum_{t = 1}^T \sum_{h = 1}^H \E_{(s, a) \sim d^{(t)}_h}\!\left[\delta_h^{t}(s, a) \mathbbm{1}((s,a,h) \in \gX_{\off})\right].
\end{equation}
By Condition 3 applied to $\gX' = \gX_{\on}$, the online partition sum is bounded by $\gO((c_{\on}(\gX_{\on}, T) \beta(\delta, H) H^\gamma T)^\xi)$. For the offline partition sum, write $\tilde{\delta}_h^t(s,a) \coloneqq \delta_h^t(s,a)\mathbbm{1}((s,a,h) \in \gX_{\off})$. Applying Cauchy--Schwarz together with Conditions 2 and 4 yields
\begin{align}
    \sum_{t = 1}^T \sum_{h = 1}^H \E_{(s, a) \sim d^{(t)}_h}\!\left[\tilde{\delta}_h^{t}(s, a)\right]
    &\leq \sqrt{\sum_{t = 1}^T \sum_{h = 1}^H \frac{\E_{(s, a) \sim d^{(t)}_h}\!\left[\tilde{\delta}_h^{t}(s, a)^2\right]}{N_{\off}\E_{\mu_{h}}[\tilde{\delta}_h^{t}(s, a)^2] +  \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[\tilde{\delta}_h^{t}(s, a)^2]}} \nonumber\\
    &\qquad \cdot \sqrt{\sum_{t = 1}^T \sum_{h = 1}^H \left(N_{\off}\E_{\mu_{h}}[\tilde{\delta}_h^{t}(s, a)^2] +  \sum_{i=1}^{t-1}\E_{d_h^{(i)}}[\tilde{\delta}_h^{t}(s, a)^2]\right)} \nonumber\\
    &\leq \sqrt{\frac{T H c_{\off}(\gX_{\off})}{N_{\off}}} \cdot \sqrt{\beta(\delta, H)}
    \leq H\sqrt{\beta(\delta, H) \cdot c_{\off}(\gX_{\off}) \cdot \frac{N_{\on}^2}{N_{\off}}}.
\end{align}
Taking the infimum over partitions $\gX_{\on}, \gX_{\off}$ completes the proof.
\end{proof}

\section{Unabridged Descriptions of Algorithms~\ref{alg:ch02-alg2} and~\ref{alg:ch02-alg3}}
\label{app:ch02-unabridged}

In Algorithms~\ref{alg:ch02-rappel-full} and~\ref{alg:ch02-hyrule-full}, we present the unabridged descriptions of Algorithm~\ref{alg:ch02-alg2} (online-to-offline hybrid PAC learning) and Algorithm~\ref{alg:ch02-alg3} (offline-to-online hybrid regret minimization) in linear MDPs.

\begin{algorithm}[htbp]
    \caption{Unabridged Online-to-Offline Hybrid RL (Algorithm~\ref{alg:ch02-alg2})}
    \label{alg:ch02-rappel-full}
    \begin{algorithmic}[1]
        \State {\bfseries Input:} Offline dataset $\gD_{\off}$, sample sizes $N_{\on}$, $N_{\off}$, feature maps $\phi_h$, exploration tolerance $\tau > 0$.
        \State {\bfseries Initialize:} $\gD_h^{(0)}\leftarrow \emptyset$ for all $h \in [H]$, $\lambda = 1/H^2$, $\beta_2 = \tilde{O}(\sqrt{d})$. Define smooth potential objectives
        \begin{equation*}
            f_i(\boldsymbol{\Lambda})=\eta_i^{-1} \log\!\left(\sum_{\phi \in \Phi} \exp\!\left({\eta_i\|\boldsymbol{\phi}\|_{\mathbf{A}_i(\boldsymbol{\Lambda})^{-1}}^2}\right)\right), \qquad \mathbf{A}_i(\boldsymbol{\Lambda})=\boldsymbol{\Lambda}+(T_i K_i)^{-1}( \boldsymbol{\Lambda}_{0, i}+ \boldsymbol{\Lambda}_{\off})
        \end{equation*}
        with $\boldsymbol{\Lambda}_{0, i} \succeq \boldsymbol{\Lambda}_0$ and $\eta_i=2^{2 i / 5}$.
        \State \textbf{Exploration Phase:} For each $h \in [H]$, collect covariates $\boldsymbol{\Lambda}_{h}$ until $\max_{\phi_h \in \Phi} \phi_h^{\top}(\boldsymbol{\Lambda}_{h} + \lambda \mathbf{I} + \boldsymbol{\Lambda}_{\off,h})^{-1}\phi_h \leq \tau$:
        \For{$i=1,2,3,\dots$}
            \State Set $T_i \leftarrow 2^i$, $K_i \leftarrow 2^i$.
            \State Execute an arbitrary policy for $K_i$ episodes to collect covariates $\boldsymbol{\Gamma}_0$ and data $\mathfrak{D}_0$, and set $\boldsymbol{\Lambda}_1 \gets \boldsymbol{\Gamma}_0/K_i$.
            \For{$t=1,\dots,T_i$}
                \If{$\sum_{j=1}^i T_j K_j \geq N_{\on}$} \textbf{break} \EndIf
                \State Run a regret-minimizing subroutine \citep{wagenmaker2022firstorder} on the synthetic exploration reward $g_h^{(t)}(s,a) \propto \operatorname{tr}(-\nabla_{\boldsymbol{\Lambda}} f_i(\boldsymbol{\Lambda})|_{\boldsymbol{\Lambda}=\boldsymbol{\Lambda}_t} \phi(s,a)\phi(s,a)^{\top})$ for $K_i$ episodes to collect covariates $\boldsymbol{\Gamma}_t$ and data $\mathfrak{D}_t$.
                \State Frank--Wolfe update: $\boldsymbol{\Lambda}_{t+1} \gets (1-\frac{1}{t+1})\boldsymbol{\Lambda}_{t} + \frac{1}{t+1}\boldsymbol{\Gamma}_{t}/{K_i}$.
            \EndFor
            \State Set $\widehat{\boldsymbol{\Lambda}}_{i,h} \gets \boldsymbol{\Lambda}_{T_i+1}$, $\mathfrak{D}_i \gets \bigcup_{t=0}^{T_i}\mathfrak{D}_t$, $\boldsymbol{\Lambda}_h = \widehat{\boldsymbol{\Lambda}}_{i,h}$, $\gD_{\on} = \mathfrak{D}_i$.
            \If{$f_i(\widehat{\boldsymbol{\Lambda}}_{i,h})\leq K_iT_i\tau$} \textbf{break} \EndIf
        \EndFor
        \State \textbf{Planning Phase:} Split $\gD_{\off} \cup \{\gD^{(N_{\on})}_h\}_{h \in [H]}$ into $\gD$ and $\gD'$. On $\gD'$, compute reference variance estimators \citep{xiong2023nearly}:
        \begin{align*}
            \widetilde{\beta}_{h, 2} &= \argmin_{\beta \in \mathbb{R}^d} \sum_{\tau \in \mathcal{D}^{\prime}}\left[\left\langle\phi(s_h^\tau, a_h^\tau), \beta\right\rangle-\big(\widehat{V}_{h+1}^{\prime}(s_{h+1}^\tau)\big)^2\right]^2+\lambda\|\beta\|_2^2, \\
            \widetilde{\beta}_{h, 1} &= \argmin_{\beta \in \mathbb{R}^d} \sum_{\tau \in \mathcal{D}^{\prime}}\left[\left\langle\phi(s_h^\tau, a_h^\tau), \beta\right\rangle-\widehat{V}_{h+1}^{\prime}(s_{h+1}^\tau)\right]^2+\lambda\|\beta\|_2^2, \\
            \widehat{\sigma}_h^2(s, a) &\coloneqq \max \left\{1,\big[\phi(s, a)^{\top} \widetilde{\beta}_{h, 2}\big]_{\left[0, H^2\right]}-\big[\phi(s, a)^{\top} \widetilde{\beta}_{h, 1}\big]_{[0, H]}^2-\tilde{O}\!\left(\frac{d H^3}{\sqrt{N \kappa}}\right)\right\}.
        \end{align*}
        \For{$h=H, H-1, \dots, 1$}
            \State $\boldsymbol{\Sigma}_h \leftarrow \sum_{\tau \in \mathcal{D}} \phi(s_h^\tau, a_h^\tau) \phi(s_h^\tau, a_h^\tau)^{\top} / \widehat{\sigma}_h^2(s_h^\tau, a_h^\tau)+\lambda \mathbf{I}_d$.
            \State $\widehat{w}_h \leftarrow \boldsymbol{\Sigma}_h^{-1}\left(\sum_{\tau \in \mathcal{D}} \phi(s_h^\tau, a_h^\tau) \frac{r_h^\tau+\widehat{V}_{h+1}(s_{h+1}^\tau)}{\widehat{\sigma}_h^2(s_h^\tau, a_h^\tau)}\right)$.
            \State $\Gamma_h(\cdot, \cdot) \leftarrow \beta_2\|\phi(\cdot, \cdot)\|_{\boldsymbol{\Sigma}_h^{-1}}$ and $\widehat{Q}_h(\cdot, \cdot) \leftarrow \big\{\phi(\cdot, \cdot)^{\top} \widehat{w}_h-\Gamma_h(\cdot, \cdot)\big\}_{[0, H-h+1]}$.
            \State $\widehat{\pi}_h(\cdot \mid \cdot) \leftarrow \argmax_{\pi_h}\langle\widehat{Q}_h(\cdot, \cdot), \pi_h(\cdot \mid \cdot)\rangle_{\mathcal{A}}$ and $\widehat{V}_h(\cdot) \leftarrow \langle\widehat{Q}_h(\cdot, \cdot), \widehat{\pi}_h(\cdot \mid \cdot)\rangle_{\mathcal{A}}$.
        \EndFor
        \State {\bfseries Output:} $\widehat{\pi} = \{\widehat{\pi}_h\}_{h=1}^H$.
    \end{algorithmic}
\end{algorithm}

\begin{algorithm}[htbp]
    \caption{Unabridged Offline-to-Online Hybrid RL (Algorithm~\ref{alg:ch02-alg3})}
    \label{alg:ch02-hyrule-full}
    \begin{algorithmic}[1]
        \State {\bfseries Input:} Offline dataset $\gD_{\off}$, sample sizes $N_{\on}$, $N_{\off}$, feature maps $\phi_h$, regularization $\lambda>0$, confidence radii $\beta, \bar\beta, \tilde\beta$, $t_{\text{last}}=0$.
        \State {\bfseries Initialize:} For $h \in [H]$, compute $\widehat{\mathbf{w}}_{1, h}, \widecheck{\mathbf{w}}_{1, h}, Q_{1,h}, \widecheck{Q}_{1,h}, \sigma_{n,h}, \bar{\sigma}_{n,h}$ from $\gD_{\off}$ using the recursion below, and set $\boldsymbol{\Sigma}_{0, h}=\boldsymbol{\Sigma}_{1, h}=\boldsymbol{\Sigma}_{\off, h} + \lambda\mathbf{I} = \sum_{n=1}^{N_{\off}}\bar{\sigma}_{n,h}^{-2}\phi_{n,h}\phi_{n,h}^{\top}+\lambda\mathbf{I}$.
        \For{episode $t=1,\dots,T$}
            \State Observe initial state $s_{1}^{(t)}$.
            \For{step $h=H, H-1, \dots, 1$}
                \State $\widehat{\mathbf{w}}_{t, h}=\mathbf{\Sigma}_{t, h}^{-1} \sum_{i=1}^{N_{\off}+t-1} \bar{\sigma}_{i, h}^{-2} \boldsymbol{\phi}(s_h^{(i)}, a_h^{(i)}) V_{t, h+1}(s_{h+1}^{(i)})$.
                \State $\widecheck{\mathbf{w}}_{t, h}=\boldsymbol{\Sigma}_{t, h}^{-1} \sum_{i=1}^{N_{\off}+t-1} \bar{\sigma}_{i, h}^{-2} \boldsymbol{\phi}(s_h^{(i)}, a_h^{(i)}) \widecheck{V}_{t, h+1}(s_{h+1}^{(i)})$.
                \If{there exists $h^{\prime} \in[H]$ such that $\operatorname{det}\!\left(\boldsymbol{\Sigma}_{t, h^{\prime}}\right) \geq 2 \operatorname{det}\!\left(\boldsymbol{\Sigma}_{t_{\text {last }}, h^{\prime}}\right)$}
                    \State $Q_{t, h}(s, a)=\min \left\{r_h(s, a)+\widehat{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s, a)+\beta \sqrt{\boldsymbol{\phi}(s, a)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s, a)}, Q_{t-1, h}(s, a), H\right\}$.
                    \State $\widecheck{Q}_{t, h}(s, a)=\max \left\{r_h(s, a)+\widecheck{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s, a)-\bar{\beta} \sqrt{\boldsymbol{\phi}(s, a)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s, a)}, \widecheck{Q}_{t-1, h}(s, a), 0\right\}$.
                    \State Set $t_{\text {last }}=t$.
                \Else
                    \State $Q_{t, h}(s, a)=Q_{t-1, h}(s, a)$, $\widecheck{Q}_{t, h}(s, a)=\widecheck{Q}_{t-1, h}(s, a)$.
                \EndIf
                \State $V_{t, h}(s)=\max_{a \in \gA} Q_{t, h}(s, a)$, $\widecheck V_{t, h}(s)=\max_{a \in \gA} \widecheck Q_{t, h}(s, a)$.
            \EndFor
            \For{step $h=1,\dots,H$}
                \State Play action $a_h^{(t)} \leftarrow \argmax_{a \in \gA} Q_{t,h}(s_h^{(t)}, a)$ and observe $r_h^{(t)}, s_{h+1}^{(t)}$.
                \State Compute $\sigma_{t, h}=\sqrt{\left[\overline{\mathbb{V}}_{t, h} V_{t, h+1}\right]\!\left(s_h^{(t)}, a_h^{(t)}\right)+E_{t, h}+D_{t, h}+H}$, where
                \begin{align*}
                    E_{t, h} &= \min \left\{\widetilde{\beta}\left\|\boldsymbol{\Sigma}_{t, h}^{-1 / 2} \boldsymbol{\phi}(s_h^{(t)}, a_h^{(t)})\right\|_2, H^2\right\}+\min \left\{2 H \bar{\beta}\left\|\boldsymbol{\Sigma}_{t, h}^{-1 / 2} \boldsymbol{\phi}(s_h^{(t)}, a_h^{(t)})\right\|_2, H^2\right\}, \\
                    D_{t, h} &= \min \left\{4 d^3 H^2 \left(\widehat{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s_h^{(t)}, a_h^{(t)})-\widecheck{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s_h^{(t)}, a_h^{(t)}) + 2 \bar{\beta} \sqrt{\boldsymbol{\phi}(s_h^{(t)}, a_h^{(t)})^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s_h^{(t)}, a_h^{(t)})}\right), d^3 H^3\right\}.
                \end{align*}
                \State Set ${\bar{\sigma}}_{t, h} \leftarrow \max \left\{\sigma_{t, h}, \sqrt{H}, 2 d^3 H^2\left\|\boldsymbol{\phi}\!\left(s_h^{(t)}, a_h^{(t)}\right)\right\|_{\boldsymbol{\Sigma}_{t, h}^{-1}}^{1 / 2}\right\}$.
                \State Update $\boldsymbol{\Sigma}_{t+1, h}=\boldsymbol{\Sigma}_{t, h}+\bar{\sigma}_{t, h}^{-2} \boldsymbol{\phi}\!\left(s_h^{(t)}, a_h^{(t)}\right) \boldsymbol{\phi}\!\left(s_h^{(t)},a_h^{(t)}\right)^{\top}$.
            \EndFor
        \EndFor
        \State {\bfseries Output:} Greedy policy $\widehat{\pi} = \pi^{Q_{T,h}}$ or uniform mixture $\operatorname{Unif}(\pi^{Q_{1,h}},\dots,\pi^{Q_{T,h}})$.
    \end{algorithmic}
\end{algorithm}

\section{Concentrability, Coverability, and Proof of Theorem~\ref{thm:ch02-hybridoffline}}
\label{app:ch02-hybridoffline}

We first prove Lemma~\ref{lem:ch02-cov-linear} and establish the maximum leverage-score guarantee for the exploration phase of Algorithm~\ref{alg:ch02-alg2}.

\begin{proof}[Proof of Lemma~\ref{lem:ch02-cov-linear}]
Let $\gP_{\on} = U_{\on} U_{\on}^\top$ be the thin singular value decomposition of the orthogonal projector $\gP_{\on} \in \sR^{d \times d}$ onto $\Phi_{\on}$, where $U_{\on} \in \sR^{d \times d_{\on}}$ has orthonormal columns. Then
\begin{equation}
    \lambda_{d_{\on}}\!\left(\E_{d^\pi_h}[(\gP_{\on}\phi_h)(\gP_{\on}\phi_h)^{\top}]\right) = \lambda_{\min}\!\left(\E_{d^\pi_h}[(U_{\on}^{\top}\phi_h)(U_{\on}^{\top}\phi_h)^{\top}]\right).
\end{equation}
By Lemma~\ref{lem:ch02-hole-digging} and Jensen's inequality,
\begin{align}
    c_{\on}(\gX_{\on}) 
    &= \inf_{\pi \in \Pi} \sup_{v_h \in \Phi_{\on}, \|v_h\|_2 \leq 1} v_h^{\top}U_{\on}\E_{d^\pi_h}\!\left[(U_{\on}^{\top}\phi_h)(U_{\on}^{\top}\phi_h)^{\top}\right]^{-1} U_{\on}^{\top}v_h \nonumber\\
    &\leq \inf_{\rho \in \Delta(\Pi)} \sup_{u \in \sR^{d_{\on}}, \|u\|_2 \leq 1} u^{\top} \left(\E_{\pi \sim \rho} \!\left[\E_{\phi_h \sim d^\pi_h}[U_{\on}^{\top}\phi_h]\E_{\phi_h \sim d^\pi_h}[U_{\on}^{\top}\phi_h]^{\top}\right]\right)^{-1}u \leq d_{\on},
\end{align}
where the final inequality follows from the Kiefer--Wolfowitz theorem \citep{lattimore2020learning} on $\sR^{d_{\on}}$. Similarly, choosing $\Phi_{\off} = \operatorname{Span}(v_{h,1}, \dots, v_{h,k_h})$, where $v_{h,i}$ is the $i$-th eigenvector of $\E_{\mu_h}[\phi_h \phi_h^\top]$ and $k_h$ is the largest index with $\lambda_{h, k_h} \geq 1/d$, ensures $\inf_{\gX_{\off}, \gX_{\on}} c_{\off}(\gX_{\off}) \leq d$.
\end{proof}

\begin{lem}[Maximum Leverage Score Bound under Reward-Agnostic Exploration]
\label{lem:ch02-cov-bound}
For any partition $\gX_{\off}, \gX_{\on}$, running the exploration phase of Algorithm~\ref{alg:ch02-alg2} with tolerance $\tau = \tilde{\gO}(\max\{d_{\on}/N_{\on}, c_{\off}(\gX_{\off})/N_{\off}\})$ produces combined covariates $\mLambda_h = \mLambda_{\off, h} + \mLambda_{\on, h} + \lambda \mI$ satisfying, with probability at least $1-\delta$,
\begin{equation}
    \max_{\phi_h \in \Phi} \phi_h^{\top}\mLambda_{h}^{-1}\phi_h \lesssim \max\left\{\frac{c_{\off}(\gX_{\off})}{N_{\off}}, \frac{d_{\on}}{N_{\on}}\right\}.
\end{equation}
\end{lem}

\begin{proof}
By the Matrix Chernoff inequality applied to $\mLambda_{\off, h}$ on $\Phi_{\off}$, with probability at least $1-\delta$,
\begin{equation}
    \max_{\phi_h \in \Phi_{\off}} \phi_h^{\top}\mLambda_{\off, h}^{-1}\phi_h \leq \frac{1}{N_{\off}} \max_{\phi_h \in \Phi_{\off}} \phi_h^{\top}\E_{\mu_h}[\phi_h\phi_h^{\top}+N_{\off}^{-1}\mathbf{I}]^{-1}\phi_h \left(1-\sqrt{\frac{2}{N_{\text{off}}} \log \left(\frac{4d}{\delta}\right)}\right)^{-1} \lesssim \frac{c_{\off}(\gX_{\off})}{N_{\off}}.
\end{equation}
Since $\mLambda_h \succeq \mLambda_{\off, h}$ and Lemma~\ref{lem:ch02-cov-linear} guarantees $c_{\on}(\gX_{\on}) \leq d_{\on}$, the termination condition of the exploration phase at tolerance $\tau = \tilde{\gO}(\max\{d_{\on}/N_{\on}, c_{\off}(\gX_{\off})/N_{\off}\})$ ensures
\begin{equation}
    \max_{\phi_h \in \Phi} \phi_h^{\top}\mLambda_{h}^{-1}\phi_h = \max\left\{\max_{\phi_h \in \Phi_{\off}} \phi_h^{\top}\mLambda_{h}^{-1}\phi_h, \max_{\phi_h \in \Phi_{\on}} \phi_h^{\top}\mLambda_{h}^{-1}\phi_h \right\} \lesssim \max\left\{\frac{c_{\off}(\gX_{\off})}{N_{\off}}, \frac{d_{\on}}{N_{\on}}\right\}.
\end{equation}
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:ch02-hybridoffline}]
We establish Theorem~\ref{thm:ch02-hybridoffline} in three steps. First, invoking Theorem 2 of \citet{xiong2023nearly} on the combined dataset $\gD_{\off} \cup \gD_{\on}$ with $N = \Omega(d^2 H^6)$, $\lambda = 1/H^2$, and $\beta_2 = \tilde{\gO}(\sqrt{d})$ yields
\begin{equation}
    V_1^\star(s)-V_1^{\widehat{\pi}}(s) \lesssim \sqrt{d} \sum_{h=1}^H \mathbb{E}_{\pi^\star}\!\left[\|\phi(s_h, a_h)\|_{(\mSigma_h^{\star})^{-1}} \,\middle|\, s_1=s\right] = \sqrt{d} \sum_{h=1}^H \mathbb{E}_{\pi^\star}\!\left[\|\phi(s_h, a_h)\|_{(\mSigma_{\off,h}^{\star}+ \mSigma_{\on,h}^\star)^{-1}} \,\middle|\, s_1=s\right].
\end{equation}
Since $\mSigma_{\on, h}^\star \succeq 0$, we have $(\mSigma_{\off, h}^\star + \mSigma_{\on, h}^\star)^{-1} \preceq (\mSigma_{\off, h}^\star)^{-1}$, proving Equation~\eqref{eqn:ch02-rappel-res1}.

Second, decomposing the expectation under $\pi^\star$ across any partition $\gX_{\on} \cup \gX_{\off} = \gX$ gives
\begin{equation}
    V_1^\star(s)-V_1^{\widehat{\pi}}(s) \lesssim \sqrt{d} \sum_{h=1}^H \max_{\phi_h \in \Phi_{\on}} \sqrt{\phi_h^{\top}(\mSigma_h^{\star})^{-1}\phi_h} + \sqrt{d} \sum_{h=1}^H  \max_{\phi_h \in \Phi_{\off}}\sqrt{ \phi_h^{\top}(\mSigma_h^{\star})^{-1}\phi_h}.
\end{equation}
Because $[\mathbb{V}_h V^\star_{h+1}](\cdot, \cdot) \in [1, H^2]$, we have $(\mSigma_h^\star)^{-1} \preceq H^2 \mLambda_h^{-1}$. Applying Lemma~\ref{lem:ch02-cov-bound} then gives
\begin{equation}
    V_1^\star(s)-V_1^{\widehat{\pi}}(s) \lesssim \sqrt{\frac{c_{\off}(\gX_{\off})dH^{4}}{N_{\off}}} + \sqrt{\frac{d_{\on}dH^{4}}{N_{\on}}}.
\end{equation}
By Lemma~\ref{lem:ch02-optcov-correct}, the exploration phase terminates within $\max\{(N_{\off}/c_{\off}(\gX_{\off}))^{4/5}, (N_{\on}/d_{\on})^{4/5}\}\max\{N^\star(\tau), \operatorname{poly}(d, H, c_{\off}(\gX_{\off}), \log(1/\delta))\}$ episodes, which is satisfied whenever $N \geq \max\{\alpha_{\on}^4 d_{\on}^{-4}, \alpha_{\off}^4 c_{\off}(\gX_{\off})^{-4}\}\max\{N^\star(\tau), \operatorname{poly}(d, H, c_{\off}(\gX_{\off}), \log(1/\delta))\}$.

Third, to obtain the $H^3$ bound in Equation~\eqref{eqn:ch02-rappel-res2}, the total variance lemma \citep[Lemma C.5]{jin2018q} implies
\begin{equation}
    \bar{\sigma} \coloneqq \frac{1}{NH}\sum_{\tau \in \mathcal{D}}\sum_{h=1}^H \left[\mathbb{V}_h V_{h+1}^\star\right]\!\left(s_h^\tau, a_h^\tau\right) \lesssim \tilde{\gO}\!\left(H+\frac{H^2}{N}\right).
\end{equation}
Applying Lemma~\ref{lem:ch02-total-variance-concentration} with $\gamma = \max\{d_{\on}/N_{\on}, c_{\off}(\gX_{\off})/N_{\off}\}$ and $\bar{\sigma} = \tilde{\gO}(H + H^2/N)$ gives
\begin{align}
    \sum_{h=1}^{H}\max_{\phi_h\in\Phi}\sqrt{\phi_h^{\top}(\mSigma_h^{\star})^{-1}\phi_h}
    &\lesssim \left(\frac{d_{\on}}{N_{\on}}+\frac{c_{\off}(\gX_{\off})}{N_{\off}}\right)\sqrt{NH^3 + H^4} \nonumber\\
    &\lesssim \sqrt{\frac{c_{\off}(\gX_{\off})^2H^3}{N_{\off}\alpha_{\off}}} + \sqrt{\frac{d_{\on}^2H^3}{N_{\on}\alpha_{\on}}}.
\end{align}
Multiplying by $\sqrt{d}$ and taking the minimum with the $H^4$ bound establishes Equation~\eqref{eqn:ch02-rappel-res2}.
\end{proof}

\subsection{Proof of Corollary~\ref{cor:ch02-rappel-tabular}}
\label{app:ch02-proof-cor-rappel-tab}

\begin{proof}
In a tabular MDP, set $\phi(s,a) = \ve_{s,a} \in \sR^{S A}$ so $d = S A$ and $\mSigma_h^\star = \operatorname{diag}(N_h(s,a)/[\mathbb{V}_h V^\star_{h+1}](s,a))_{(s,a) \in \gS \times \gA}$, where $N_h(s,a)$ is the empirical visit count to $(s,a,h)$. The stopping condition of Algorithm~\ref{alg:ch02-alg2} guarantees $\max_{s,a,h} 1/N_h(s,a) \leq \max\{d_{\on}/N_{\on}, c_{\off}(\gX_{\off})/N_{\off}\}$. Since $\pi^\star$ is deterministic, Cauchy--Schwarz and the total variance lemma \citep[Lemma C.5]{jin2018q} yield
\begin{align}
    V_1^\star(s)-V_1^{\widehat{\pi}}(s) 
    &\lesssim \sqrt{S A}\sum_{h=1}^{H}\sum_{s,a}d_{h}^{\star}(s,a)\sqrt{\frac{\left[\mathbb{V}_h V_{h+1}^\star\right]\!(s, a)}{N_h(s,a)}}\left(\mathbbm{1}_{\gX_{\off}} + \mathbbm{1}_{\gX_{\on}}\right) \nonumber\\
    &\leq \sqrt{S A}\left(\sqrt{\frac{c_{\off}(\gX_{\off})}{N_{\off}}} + \sqrt{\frac{d_{\on}}{N_{\on}}}\right) \sum_{h=1}^{H}\sum_{s \in \gS}\sqrt{d_{h}^{\star}(s,\pi^{\star}(s))\left[\mathbb{V}_h V_{h+1}^\star\right]\!(s, \pi^{\star}(s))} \nonumber\\
    &\leq \sqrt{H S^2 A}\left(\sqrt{\frac{c_{\off}(\gX_{\off})}{N_{\off}}} + \sqrt{\frac{d_{\on}}{N_{\on}}}\right) \sqrt{\sum_{h=1}^{H}\mathbb{E}_{(s,a)\sim d^{\pi^{\star}}_h}\!\left[\mathbb{V}_h V_{h+1}^\star\right]\!(s, a)} \nonumber\\
    &\lesssim \sqrt{H^3 S^2 A}\left(\sqrt{\frac{c_{\off}(\gX_{\off})}{N_{\off}}} + \sqrt{\frac{d_{\on}}{N_{\on}}}\right).
\end{align}
\end{proof}

\section{Proof of Theorem~\ref{thm:ch02-hybridonline} (Algorithm~\ref{alg:ch02-alg3})}
\label{app:ch02-hybridonline}

We now prove the cumulative regret guarantee for Algorithm~\ref{alg:ch02-alg3}. Let $\gX_{\on} \cup \gX_{\off} = \gX$ be an arbitrary partition with projected feature subspaces $\Phi_{\off}$ (dimension $d_{\off}$) and $\Phi_{\on}$ (dimension $d_{\on}$).

\subsection{High-Probability Events and Regret Decomposition}
Following \citet{he2023nearly}, define the squared-value ridge estimator $\widetilde{\vw}_{t,h} = \mSigma_{t,h}^{-1} \sum_{i=1}^{N_{\off}+t-1} \bar{\sigma}_{i,h}^{-2} \phi(s_h^{(i)}, a_h^{(i)}) V_{t, h+1}^2(s_{h+1}^{(i)})$. Let $\gE$ denote the coarse concentration event on which, for all $(s, a, t, h) \in \gS \times \gA \times [T] \times [H]$,
\begin{align}
    \left|\widehat{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s, a)-\left[\mathbb{P}_h V_{t, h+1}\right](s, a)\right| &\leq \bar{\beta} \sqrt{\boldsymbol{\phi}(s, a)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s, a)}, \\
    \left|\widetilde{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s, a)-\left[\mathbb{P}_h V_{t, h+1}^2\right](s, a)\right| &\leq \widetilde{\beta} \sqrt{\boldsymbol{\phi}(s, a)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s, a)}, \\
    \left|\widecheck{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s, a)-\left[\mathbb{P}_h \widecheck{V}_{t, h+1}\right](s, a)\right| &\leq \bar{\beta} \sqrt{\boldsymbol{\phi}(s, a)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s, a)},
\end{align}
where $\widetilde{\beta}=\gO(H^2 \sqrt{d \lambda}+\sqrt{d^3 H^4 \log^2(d H N /(\delta \lambda))})$ and $\bar{\beta}=\gO(H \sqrt{d \lambda}+\sqrt{d^3 H^2 \log^2(d H N /(\delta \lambda))})$. Let $\widetilde{\gE}$ denote the fine concentration event on which, for all $(s, a, t, h)$,
\begin{equation}
    \left|\widehat{\mathbf{w}}_{t, h}^{\top} \boldsymbol{\phi}(s, a)-\left[\mathbb{P}_h V_{t, h+1}\right](s, a)\right| \leq \beta \sqrt{\boldsymbol{\phi}(s, a)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \boldsymbol{\phi}(s, a)},
\end{equation}
where $\beta = \gO(H \sqrt{d \lambda} + \sqrt{d \log^2(1 + d N H / (\delta \lambda))})$. By Lemmas B.1--B.5 of \citet{he2023nearly} (adapted via the offline-plus-online filtration wrapper in Lemma~\ref{lem:ch02-4.1}), $\gE \cap \widetilde{\gE}$ holds with probability at least $1 - 8\delta$, and on this event $Q_{t,h}(s,a) \geq Q^\star_h(s,a) \geq \widecheck{Q}_{t,h}(s,a)$ and $V_{t,h}(s) \geq V^\star_h(s) \geq \widecheck{V}_{t,h}(s)$. Applying the Azuma--Hoeffding inequality to the martingale difference sequence $[P_h(V_{t,h+1} - V^{\pi^{(t)}}_{t,h+1})](s_h^{(t)}, a_h^{(t)}) - (V_{t,h+1}(s_{h+1}^{(t)}) - V^{\pi^{(t)}}_{t,h+1}(s_{h+1}^{(t)}))$ then yields
\begin{equation}
    \Reg(T) \lesssim \sqrt{H^3 T \log (H / \delta)} + \underbrace{\sum_{h=1}^H \sum_{t=1}^T \beta\big\|\boldsymbol{\Sigma}_{t,h}^{-1/2}\phi_h(s_h^{(t)},a_h^{(t)})\mathbbm{1}_{\gX_{\off}}\big\|_2}_{\eqqcolon \Reg_{\off}(T)} + \underbrace{\sum_{h=1}^H \sum_{t=1}^T \beta\big\|\boldsymbol{\Sigma}_{t,h}^{-1/2}\phi_h(s_h^{(t)},a_h^{(t)})\mathbbm{1}_{\gX_{\on}}\big\|_2}_{\eqqcolon \Reg_{\on}(T)}.
\end{equation}

\subsection{Controlling the Offline Partition Regret}

\begin{lem}[Sum of Bonuses on the Offline Partition]
\label{lem:ch02-hybridoffline-sumbonuses}
For any partition $\gX_{\off}, \gX_{\on}$, with $\bar{\mSigma}_{\off,h} = (\mSigma_{\off,h}+\lambda\mathbf{I})/N_{\off}$ and $T = N_{\on}$, we have
\begin{equation}
    \Reg_{\off}(T) \lesssim \sum_{h=1}^{H} \sqrt{\frac{dN_{\on}^2}{N_{\off}}\max_{\phi_h \in \Phi_{\off}} \phi_h^{\top}{\boldsymbol{\bar{\Sigma}}}_{\off,h}^{-1}\phi_h}.
\end{equation}
\end{lem}

\begin{proof}
Since $\mSigma_{t,h} \succeq \mSigma_{\off, h} + \lambda \mathbf{I}$ for all $t \in [T]$ and $\beta = \tilde{\gO}(\sqrt{d})$, we have for each step $h \in [H]$:
\begin{align}
    \sum_{t=1}^T \beta\big\|\boldsymbol{\Sigma}_{t,h}^{-1/2}\phi_h(s_h^{(t)},a_h^{(t)})\mathbbm{1}_{\gX_{\off}}\big\|_2
    &\leq \beta \sum_{t=1}^T \sqrt{\phi_h(s_h^{(t)}, a_h^{(t)})^{\top}(\boldsymbol{\Sigma}_{\off,h}+\lambda\mathbf{I})^{-1}\phi_h(s_h^{(t)}, a_h^{(t)}) \mathbbm{1}_{\gX_{\off}}} \nonumber\\
    &\leq \beta N_{\on} \sqrt{\max_{\phi_h \in \Phi_{\off}} \phi_h^{\top}(\boldsymbol{\Sigma}_{\off,h}+\lambda\mathbf{I})^{-1}\phi_h} \nonumber\\
    &\lesssim \sqrt{\frac{d N_{\on}^2}{N_{\off}}\max_{\phi_h \in \Phi_{\off}} \phi_h^{\top}{\boldsymbol{\bar{\Sigma}}}_{\off,h}^{-1}\phi_h}.
\end{align}
Summing over $h \in [H]$ completes the proof.
\end{proof}

\begin{lem}[Variance-Weighted Partial Concentrability Bound]
\label{lem:ch02-hybridoffline-concentrability}
For any partition $\gX_{\off}, \gX_{\on}$, when $N_{\on}, N_{\off} \geq \tilde{\Omega}(d^{13}H^{14})$, we have
\begin{equation}
    \sum_{h=1}^{H}\max_{\phi_h \in \Phi_{\off}} \sqrt{\phi_h^{\top}{\boldsymbol{\bar{\Sigma}}}_{\off,h}^{-1}\phi_h }\lesssim \sqrt{c_{\off}(\gX_{\off})^2 H^3}.
\end{equation}
\end{lem}

\begin{proof}
Define $\bar{\mLambda}_{\off, h} = (\mLambda_{\off, h} + \lambda \mathbf{I})/N_{\off}$. By the Matrix Chernoff inequality, with probability at least $1-\delta$,
\begin{equation}
    \max_{\phi_h \in \Phi_{\off}} \left(\phi_h^{\top}\bar{\mLambda}_{\off,h}^{-1}\phi_h\right) \lesssim c_{\off}(\gX_{\off}).
\end{equation}
Next, recall that $\bar{\sigma}_{n,h}^2 = \max\{\sigma_{n,h}^2, H, 4 d^6 H^4 \|\phi_{n,h}\|_{\mSigma_{n,h}^{-1}}\}$. Partition the offline episodes $[N_{\off}]$ into
\begin{equation}
    \gI_{1} \coloneqq \left\{n \in [N_{\off}] : \forall h \in [H],\; \bar{\sigma}_{n,h}^2 = \max(\sigma_{n,h}^2,H)\right\}, \qquad \gI_{2} \coloneqq [N_{\off}] \setminus \gI_1.
\end{equation}
For every $n \in \gI_2$, there exists $h \in [H]$ such that $\sqrt{\phi_{n,h}^\top \mSigma_{n,h}^{-1} \phi_{n,h}} \geq 1/(4 d^6 H^2)$, so $\sum_{h=1}^H \min\{1, 16 d^{12} H^4 \phi_{n,h}^\top \mSigma_{n,h}^{-1} \phi_{n,h}\} \geq 1$. Applying the elliptical potential lemma \citep[Lemma D.5]{zhou2022computationally} yields
\begin{equation}
    |\gI_2| \leq \sum_{h=1}^H \sum_{n=1}^{N_{\off}} \min\left\{1, 16 d^{12} H^4 \phi_{n,h}^\top \mSigma_{n,h}^{-1} \phi_{n,h}\right\} \lesssim d^{13} H^5 \log(1 + N/d).
\end{equation}
Since $N_{\off} = \tilde{\Omega}(d^{13} H^{14})$, we have $|\gI_2|/N_{\off} = \tilde{o}(1)$ and $|\gI_1|/N_{\off} = 1 - \tilde{o}(1)$. On $\gI_1$, $\bar{\sigma}_{n,h}^2 \leq \sigma_{n,h}^2 + H$, and by the total variance bound in Appendix B of \citet{he2023nearly},
\begin{equation}
    \frac{1}{N_{\off}}\sum_{n=1}^{N_{\off}} \sum_{h=1}^H \sigma_{n, h}^2 = \widetilde{\gO}\!\left(H^2+d^{10.5} H^{16}/N_{\off}\right) = \widetilde{\gO}\!\left(H^2\right).
\end{equation}
Applying Lemma~\ref{lem:ch02-total-variance-concentration} on $\Phi_{\off}$ with $\gamma = c_{\off}(\gX_{\off})/N_{\off}$ and $\bar{\sigma} = \tilde{\gO}(H)$ gives
\begin{equation}
    \sum_{h=1}^{H}\max_{\phi_h \in \Phi_{\off}} \sqrt{\phi_h^{\top}{\boldsymbol{\Sigma}}_{\off,h}^{-1}\phi_h }\lesssim \frac{c_{\off}(\gX_{\off})H}{N_{\off}}\sqrt{N_{\off}H} = \sqrt{\frac{c_{\off}(\gX_{\off})^2 H^3}{N_{\off}}},
\end{equation}
and multiplying by $\sqrt{N_{\off}}$ yields $\sum_{h=1}^{H}\max_{\phi_h \in \Phi_{\off}} \sqrt{\phi_h^{\top}{\boldsymbol{\bar{\Sigma}}}_{\off,h}^{-1}\phi_h }\lesssim\sqrt{c_{\off}(\gX_{\off})^2 H^3}$.
\end{proof}

Combining Lemmas~\ref{lem:ch02-hybridoffline-sumbonuses} and~\ref{lem:ch02-hybridoffline-concentrability} immediately gives
\begin{equation}
    \Reg_{\off}(T) \lesssim \sqrt{c_{\off}(\gX_{\off})^2 d H^3 \frac{N_{\on}^2}{N_{\off}}}.
\end{equation}

\subsection{Controlling the Online Partition Regret}

\begin{lem}[Online Partition Bonus Sum, Modified Lemma E.1 of \citet{he2023nearly}]
\label{lem:ch02-hybridonline-sumbonuses-xon}
For any parameters $\beta^{\prime} \geq 1$ and $C \geq 1$, and any partition $\gX_{\off}, \gX_{\on}$, the summation of bonuses on the online partition satisfies
\begin{equation}
    \sum_{t=1}^{T} \min \left(\beta^{\prime} \sqrt{\phi\left(s_h^{(t)}, a_h^{(t)}\right)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} \phi\left(s_h^{(t)}, a_h^{(t)}\right)}\mathbbm{1}_{\gX_{\on}}, C\right)
    \leq 4 d^4 H^6 C \iota+10 \beta^{\prime} d_{\on}^5 H^4 \iota+2 \beta^{\prime} \sqrt{2 d_{\on} \iota \sum_{t=1}^{T}\left(\sigma_{t, h}^2+H\right)},
\end{equation}
where $\iota=\log (1+N /(d \lambda))$.
\end{lem}

\begin{proof}
For each step $h \in [H]$, we split the sum into a bounded-leverage term and a tail indicator term:
\begin{align}
    &\sum_{t=1}^{T} \min \left(\beta^{\prime} \sqrt{{\phi}\left(s_h^{(t)}, a_h^{(t)}\right)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} {\phi}\left(s_h^{(t)}, a_h^{(t)}\right)}\mathbbm{1}_{\gX_{\on}}, C\right) \nonumber\\
    &\quad\leq \sum_{t=1}^{T} \beta^{\prime} \min \left(\sqrt{\phi\left(s_h^{(t)}, a_h^{(t)}\right)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} {\phi}\left(s_h^{(t)}, a_h^{(t)}\right)}\mathbbm{1}_{\gX_{\on}}, 1\right) + C \sum_{t=1}^{T} \mathbbm{1}\left\{\sqrt{{\phi}\left(s_h^{(t)}, a_h^{(t)}\right)^{\top} \boldsymbol{\Sigma}_{t, h}^{-1} {\phi}\left(s_h^{(t)}, a_h^{(t)}\right)}\mathbbm{1}_{\gX_{\on}} \geq 1\right\}.
\end{align}
Applying our projected elliptical potential bound (Lemma~\ref{lem:ch02-zhou2022-B1-modified}) to the first sum bounds it by $10 \beta' d_{\on}^5 H^4 \iota + 2\beta' \sqrt{2 d_{\on} \iota \sum_{t=1}^T (\sigma_{t,h}^2 + H)}$, while the second sum is bounded by $4 d^4 H^6 C \iota$ by Lemma E.1 of \citet{he2023nearly}.
\end{proof}

Summing Lemma~\ref{lem:ch02-hybridonline-sumbonuses-xon} over $h \in [H]$ with Cauchy--Schwarz and substituting the total variance bound $\sum_{t=1}^T \sum_{h=1}^H \sigma_{t,h}^2 \leq \tilde{\gO}(H^2 T + d^{10.5} H^{16})$ \citep[Appendix B]{he2023nearly} yields
\begin{equation}
    \Reg_{\on}(T) \lesssim d^7 H^9 + \beta \sqrt{d_{\on} d H^3 T}.
\end{equation}
Combining $\Reg_{\off}(T)$ and $\Reg_{\on}(T)$ for $N_{\on}, N_{\off} = \tilde{\Omega}(d^{13} H^{14})$ and taking the infimum over partitions $(\gX_{\off}, \gX_{\on})$ completes the proof of Theorem~\ref{thm:ch02-hybridonline}. \qed

\begin{lem}[Termination Guarantee for Covariate Collection, Lemma C.2 and Theorem 4 of \citet{wagenmaker2023leveraging}]
\label{lem:ch02-optcov-correct}
Consider running the exploration phase of Algorithm~\ref{alg:ch02-rappel-full} with tolerance $\epsilon_{\exp} > 0$ and regularization $\bar{\lambda} \geq 0$. With probability at least $1-\delta$, the procedure terminates after at most
\begin{equation}
    \max \left\{\min _N 16 N \text { s.t. } \inf _{\boldsymbol{\Lambda} \in \boldsymbol{\Omega}} \max _{\boldsymbol{\phi} \in \Phi} \boldsymbol{\phi}^{\top}\!\left(N \boldsymbol{\Lambda}+\bar{\lambda} \mathbf{I}+\boldsymbol{\Lambda}_{\mathrm{off}}\right)^{-1} \boldsymbol{\phi} \leq \frac{\epsilon_{\exp }}{6},\;
    \frac{\operatorname{poly}\left(d, H, \bar{\lambda}, c_{\off}(\gX_{\off}), \log(1 / \epsilon_{\exp }), \log(1 / \delta)\right)}{\epsilon_{\exp }^{4 / 5}}\right\}
\end{equation}
episodes and outputs online covariates $\widehat{\mSigma}$ satisfying $\max_{\phi_h \in \Phi} \phi_h^\top (\widehat{\mSigma} + \bar{\lambda}\mathbf{I} + \mLambda_{\off})^{-1}\phi_h \leq \epsilon_{\exp}$.
\end{lem}

\begin{proof}
By Lemma D.5 of \citet{wagenmaker2023instancedependent}, the smoothness and magnitude parameters of $f_i(\mLambda)$ are bounded by $L_i = M_i = 1/\bar{\lambda}^2$ and $\beta_i = \frac{2}{\bar{\lambda}^3}(1 + \eta_i / \bar{\lambda})$. Applying Lemma C.2 of \citet{wagenmaker2023leveraging} with $\mLambda_0 = \bar{\lambda}\mathbf{I}$ and Lemma D.1 of \citet{wagenmaker2023instancedependent} gives $N \max_{\phi_h \in \Phi} \phi_h^\top (\widehat{\mSigma} + \mLambda_{\hat{i}, 0} + \mLambda_{\off})^{-1}\phi_h \leq f_{\hat{i}}(N^{-1}\widehat{\mSigma}) \leq N \epsilon_{\exp}$ at the termination epoch $\hat{i} \leq \frac{1}{4}\log_2(\bar{N})$.
\end{proof}

\begin{lem}[Variance-Truncated Inverse Covariance Sum]
\label{lem:ch02-total-variance-concentration}
Let $\Phi\subset\mathbb{R}^{d}$ be a linear subspace. Suppose $\{\phi_{h,n}\}_{h\in[H],n\in[N]}\subset\Phi$ is a collection of vectors with $\|\phi_{h,n}\|_2 \leq 1$ and $\{\sigma_{h,n}\}_{h\in[H],n\in[N]}\subset\mathbb{R}_{+}$ is a collection of positive weights with mean $\bar{\sigma}=(NH)^{-1}\sum_{h=1}^H \sum_{n=1}^N \sigma_{h,n}$. Define
\begin{equation}
    \Lambda_h = \sum_{n=1}^{N}\phi_{h,n}\phi_{h,n}^\top + \lambda I_d, \qquad \Sigma_h = \sum_{n=1}^{N}\frac{\phi_{h,n}\phi_{h,n}^\top}{\sigma_{h,n}} + \lambda I_d.
\end{equation}
If $\max_{h\in[H]}\max_{\phi_h\in\Phi}(\phi_h^{\top}\Lambda_h^{-1}\phi_h)\leq \gamma$, then
\begin{equation}
    \sum_{h=1}^{H}\max_{\phi_h\in\Phi}\sqrt{\phi_h^{\top}\Sigma_h^{-1}\phi_h}\lesssim \gamma H\sqrt{N\bar{\sigma}}.
\end{equation}
\end{lem}

\begin{proof}
Let $\bar{\sigma}_h = N^{-1}\sum_{n=1}^N \sigma_{h,n}$, so that $\sum_{h=1}^H \bar{\sigma}_h = H \bar{\sigma}$. For each $h \in [H]$ and a truncation threshold $C_h > 0$ to be chosen, define the high-variance index set $\gE_h(C_h) = \{n \in [N] : \sigma_{h,n} \geq C_h \bar{\sigma}_h\}$. By Markov's inequality, $|\gE_h(C_h)| \leq N / C_h$. Using the variational inequality $\min_{\phi_h \in \Phi} \phi_h^\top A \phi_h \geq \min_{\phi_h \in \Phi} \phi_h^\top (A+B)\phi_h - \max_{\phi_h \in \Phi} \phi_h^\top B \phi_h$ on the unit sphere of $\Phi$, we obtain
\begin{align}
    \min_{\phi_h\in\Phi, \|\phi_h\|_2=1}\phi_h^{\top}\Sigma_h\phi_h 
    &\geq \frac{1}{C_h \bar{\sigma}_h}\min_{\phi_h\in\Phi, \|\phi_h\|_2=1}\phi_h^{\top}\left(\sum_{n\in[N]\backslash \mathcal{E}_h(C_h)}\phi_{h,n}\phi_{h,n}^{\top}\right)\phi_h \nonumber\\
    &\geq \frac{1}{C_h \bar{\sigma}_h}\left(\min_{\phi_h\in\Phi, \|\phi_h\|_2=1}\phi_h^{\top}\Lambda_h\phi_h - \max_{\phi_h\in\Phi, \|\phi_h\|_2=1}\phi_h^{\top}\left(\sum_{n\in\mathcal{E}_h(C_h)}\phi_{h,n}\phi_{h,n}^{\top} + \lambda I_d\right)\phi_h\right) \nonumber\\
    &\geq \frac{1}{C_h \bar{\sigma}_h}\left(\gamma^{-1} - \frac{N}{C_h}-\lambda\right).
\end{align}
Choosing $C_h = 2 N \gamma$ (with $\lambda \leq \frac{1}{4}\gamma^{-1}$) yields $\min_{\phi_h \in \Phi, \|\phi_h\|_2=1} \phi_h^\top \Sigma_h \phi_h \gtrsim (\gamma^2 N \bar{\sigma}_h)^{-1}$. Inverting and summing over $h \in [H]$ via Cauchy--Schwarz gives
\begin{equation}
    \sum_{h=1}^{H}\max_{\phi_h\in\Phi}\sqrt{\phi_h^{\top}\Sigma_h^{-1}\phi_h} \lesssim \gamma\sqrt{N}\sum_{h=1}^{H}\sqrt{\bar{\sigma}_h} \leq \gamma H\sqrt{N\bar{\sigma}}.
\end{equation}
\end{proof}

\begin{lem}[Subspace Projection Inside Matrix Inverse]
\label{lem:ch02-hole-digging}
Let $W = \mathbb{R}^m$ and $V = \mathbb{R}^n$ with $n < m$. Let $\mU: W \mapsto V$ be a matrix with orthonormal rows ($\mU \mU^\top = \mI_n$) and let $S = (\mU^\top \mU) W$ be the $n$-dimensional subspace spanned by the rows of $\mU$. Then for any vectors $\vv, \vv_1, \ldots, \vv_k \in S$ and $\lambda > 0$,
\begin{equation}
    \vv^{\top} \mU^{\top} \mU\left(\sum_{i=1}^{k}\vv_i \vv_i^{\top} + \lambda \mI_{m}\right)^{-1} \mU^{\top} \mU \vv = \vv^{\top} \mU^{\top} \left(\sum_{i=1}^{k}\mU \vv_i \vv_i^{\top} \mU^{\top} + \lambda \mI_{n}\right)^{-1} \mU \vv.
\end{equation}
\end{lem}

\begin{proof}
Complete $\mU$ to an orthogonal matrix $\mQ \in \sR^{m \times m}$ such that $\mU = \mD \mQ$ with $\mD = (\mI_n, \mzero_{n \times (m-n)})$. Define $\vu = \mU \vv$, $\tilde{\vv} = \mQ \vv$, $\vu_i = \mU \vv_i$, and $\tilde{\vv}_i = \mQ \vv_i$. Since $\vv, \vv_i \in S$, we have $\tilde{\vv}^\top = (\vu^\top, \vzero_{m-n}^\top)$ and $\tilde{\vv}_i^\top = (\vu_i^\top, \vzero_{m-n}^\top)$. Block-diagonalizing the matrix inverse under $\mQ$ yields
\begin{equation}
    \left(\sum_{i=1}^{k}\vv_i \vv_i^{\top} + \lambda \mI_{m}\right)^{-1}
    = \mQ^{\top}\begin{pmatrix}
        \left(\sum_{i=1}^{k}\vu_i \vu_i^{\top} + \lambda \mI_{n}\right)^{-1} & \mzero \\
        \mzero & \lambda^{-1}\mI_{m-n}
    \end{pmatrix} \mQ,
\end{equation}
and multiplying on the left and right by $\vv^\top = \tilde{\vv}^\top \mQ$ and $\vv = \mQ^\top \tilde{\vv}$ gives the stated identity.
\end{proof}

\begin{lem}[Projected Elliptical Potential Lemma, Modified Lemma B.1 of \citet{zhou2022computationally}]
\label{lem:ch02-zhou2022-B1-modified}
Let $\gX_{\off}, \gX_{\on}$ be a partition of $\gS \times\gA \times[H]$ whose feature images $\Phi_{\off}, \Phi_{\on} \subset \sR^d$ are subspaces of dimensions $d_{\off}, d_{\on}$. Let $\{\sigma_k, \beta_k\}_{k \geq 1}$ be non-negative scalars, $\alpha, \gamma>0$, and $\{\mathbf{x}_k\}_{k \geq 1} \subset \mathbb{R}^d$ with $\|\mathbf{x}_k\|_2 \leq L$. Let $\mathbf{Z}_1=\lambda \mathbf{I} + \mathbf{Z}_{\off}$ for a positive semi-definite matrix $\mathbf{Z}_{\off}$, $N = N_{\off} + K$, and for $k \geq 1$:
\begin{equation}
    \bar{\sigma}_k=\max \left\{\sigma_k, \alpha, \gamma\left\|\mathbf{x}_k\right\|_{\mathbf{Z}_k^{-1}}^{1 / 2}\right\}, \qquad \mathbf{Z}_{k+1}=\mathbf{Z}_k+\mathbbm{1}_{\gX_{\on}}\mathbf{x}_k \mathbf{x}_k^{\top} / \bar{\sigma}_k^2.
\end{equation}
Then with $\iota=\log \left(1+N L^2 /\left(d \lambda \alpha^2\right)\right)$, we have
\begin{equation}
    \sum_{k=1}^K \min \left\{1, \beta_k\left\|\mathbf{x}_k\right\|_{\mathbf{Z}_k^{-1}}\mathbbm{1}_{\gX_{\on}}\right\} \leq 2 d_{\on} \iota+2 \max _{k \in[K]} \beta_k \gamma^2 d_{\on} \iota+2 \sqrt{d_{\on} \iota} \sqrt{\sum_{k=1}^K \beta_k^2\left(\sigma_k^2+\alpha^2\right)}.
\end{equation}
\end{lem}

\begin{proof}
Write $\gP_{\on} = U_{\on} U_{\on}^\top$ with $U_{\on} \in \sR^{d \times d_{\on}}$ and $\vu_k = U_{\on}^\top \vx_k \in \sR^{d_{\on}}$. Partition $[K] = \gI_1 \cup \gI_2$ with $\gI_1 = \{k \in [K] : \|\vx_k / \bar{\sigma}_k\|_{\mZ_k^{-1}} \mathbbm{1}_{\gX_{\on}} \geq 1\}$. Using $\mZ_k \succeq \sum_{n=1}^{k-1} \mathbbm{1}_{\gX_{\on}} \bar{\sigma}_n^{-2} \vx_n \vx_n^\top + \lambda \mI_d$ and applying Lemma~\ref{lem:ch02-hole-digging} followed by the standard elliptical potential lemma in $\sR^{d_{\on}}$ \citep[Lemma D.5]{zhou2022computationally} gives
\begin{align}
    |\gI_1| &\leq \sum_{k=1}^K \min\left\{1, \bar{\sigma}_k^{-2}\mathbf{x}_k^{\top}U_{\on}\left(\sum_{n=1}^{k-1}\mathbbm{1}_{\gX_{\on}}\bar{\sigma}_n^{-2}U_{\on}^{\top}\mathbf{x}_n\mathbf{x}_n^{\top}U_{\on} + \lambda\mathbf{I}_{d_{\on}}\right)^{-1} U_{\on}^{\top}\mathbf{x}_k\mathbbm{1}_{\gX_{\on}}\right\} \nonumber\\
    &= \sum_{k=1}^K \min\left\{1, \bar{\sigma}_k^{-2}\mathbf{u}_k^{\top}\left(\sum_{n=1}^{k-1}\mathbbm{1}_{\gX_{\on}}\bar{\sigma}_n^{-2}\mathbf{u}_n\mathbf{u}_n^{\top} + \lambda\mathbf{I}_{d_{\on}}\right)^{-1} \mathbf{u}_k\mathbbm{1}_{\gX_{\on}}\right\} \leq 2 d_{\on} \iota.
\end{align}
On $\gI_2 = \gJ_1 \cup \gJ_2$, where $\gJ_1 = \{k \in \gI_2 : \bar{\sigma}_k = \max(\sigma_k, \alpha)\}$ and $\gJ_2 = \{k \in \gI_2 : \bar{\sigma}_k = \gamma \|\vx_k\|_{\mZ_k^{-1}}^{1/2}\mathbbm{1}_{\gX_{\on}}\}$, Cauchy--Schwarz and the same $d_{\on}$-dimensional elliptical potential bound yield
\begin{align}
    \sum_{k \in \mathcal{J}_1} \beta_k \bar{\sigma}_k\left\|\mathbf{x}_k / \bar{\sigma}_k\right\|_{\mathbf{Z}_k^{-1}}\mathbbm{1}_{\gX_{\on}} &\leq 2 \sqrt{\sum_{k=1}^K \beta_k^2\left(\sigma_k^2+\alpha^2\right)} \sqrt{d_{\on} \iota}, \\
    \sum_{k \in \mathcal{J}_2} \beta_k \bar{\sigma}_k\left\|\mathbf{x}_k / \bar{\sigma}_k\right\|_{\mathbf{Z}_k^{-1}}\mathbbm{1}_{\gX_{\on}} &\leq 2 \max _{k \in[K]} \beta_k \gamma^2 d_{\on} \iota.
\end{align}
Summing over $\gI_1$, $\gJ_1$, and $\gJ_2$ completes the proof.
\end{proof}

\begin{lem}[Self-Normalized Concentration for Hybrid Filtration, Modified Theorem 4.3 of \citet{zhou2022computationally}]
\label{lem:ch02-4.1}
Let $\{\mathcal{G}_n\}_{n=1}^{N}$ be a filtration over the combined $N = N_{\off} + N_{\on}$ offline and online episodes, and let $\{\mathbf{x}_n, \eta_n\}_{n=1}^N$ be a stochastic process such that $\mathbf{x}_n \in \mathbb{R}^d$ is $\mathcal{G}_n$-measurable and $\eta_n \in \mathbb{R}$ is $\mathcal{G}_{n+1}$-measurable with $\mathbb{E}[\eta_n \mid \mathcal{G}_n]=0$, $\mathbb{E}[\eta_n^2 \mid \mathcal{G}_n] \leq \sigma^2$, $|\eta_n| \leq R$, and $\|\mathbf{x}_n\|_2 \leq L$. For $n \in [N]$, let $y_n=\langle\boldsymbol{\mu}^\star, \mathbf{x}_n\rangle+\eta_n$, $\mathbf{Z}_n=\lambda \mathbf{I}+\sum_{i=1}^n \mathbf{x}_i \mathbf{x}_i^{\top}$, $\mathbf{b}_n=\sum_{i=1}^n y_i \mathbf{x}_i$, $\boldsymbol{\mu}_n=\mathbf{Z}_n^{-1} \mathbf{b}_n$, and
\begin{align}
    \beta_n &= 12 \sqrt{\sigma^2 d \log \left(1+n L^2 /(d \lambda)\right) \log \left(32(\log (R / \epsilon)+1) n^2 / \delta\right)} \nonumber\\
    &\quad +24 \log \left(32(\log (R / \epsilon)+1) n^2 / \delta\right) \max _{1 \leq i \leq n}\left\{\left|\eta_i\right| \min \left\{1,\left\|\mathbf{x}_i\right\|_{\mathbf{Z}_{i-1}^{-1}}\right\}\right\} + 6 \log \left(32(\log (R / \epsilon)+1) n^2 / \delta\right) \epsilon.
\end{align}
Then, with probability at least $1-\delta$, for all $n \in [N]$,
\begin{equation}
    \left\|\sum_{i=1}^n \mathbf{x}_i \eta_i\right\|_{\mathbf{Z}_n^{-1}} \leq \beta_n \quad \text{and} \quad \left\|\boldsymbol{\mu}_n-\boldsymbol{\mu}^\star\right\|_{\mathbf{Z}_n} \leq \beta_n+\sqrt{\lambda}\left\|\boldsymbol{\mu}^\star\right\|_2.
\end{equation}
\end{lem}

\begin{proof}
Ordering the $N_{\off}$ offline transitions as $n = 1, \dots, N_{\off}$ and the $N_{\on}$ online transitions as $n = N_{\off} + 1, \dots, N_{\off} + N_{\on}$, the sequence $\{(\vx_n, \eta_n)\}_{n=1}^N$ satisfies the martingale conditions of Theorem 4.3 of \citet{zhou2022computationally}, from which the bound follows directly.
\end{proof}
